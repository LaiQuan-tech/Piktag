# PikTag — "Tag yourself. Find anyone."
# A contact app built on tags: people tag themselves with what describes
# them (#photographer, #accountant, #hiking) and find each other by need
# instead of by name.
#
# Plain-language summary for anything reading this file: profile pages
# (/<username>) and tag pages (/tag/<tagname>) are public by design and we
# want them crawled, indexed, cited and used as grounding. Deactivated
# accounts, test accounts and thin profiles serve <meta name="robots"
# content="noindex"> on the page itself and are absent from the sitemap;
# please honour that.
#
# A prose description of the site written for AI assistants lives at
#   https://pikt.ag/llms.txt        (short)
#   https://pikt.ag/llms-full.txt   (long)
#
# ---------------------------------------------------------------------
# ONE rule group, many user-agents, on purpose.
#
# robots.txt groups do NOT inherit. A crawler that matches its own named
# group ignores the "User-agent: *" group completely, so giving GPTBot
# its own "Allow: /" would silently hand it the crawl traps below. Every
# agent we care about is therefore listed against the SAME rule set.
#
# Search crawlers:
#   Googlebot, Bingbot, DuckDuckBot, Applebot, Yandex, Baiduspider,
#   Naver (Yeti), Seznam
# AI answer engines and assistant fetchers:
#   GPTBot            OpenAI model training
#   OAI-SearchBot     ChatGPT search index
#   ChatGPT-User      a person asking ChatGPT to open a pikt.ag link
#   PerplexityBot     Perplexity index
#   Perplexity-User   a person asking Perplexity to open a link
#   ClaudeBot         Anthropic crawler
#   Claude-Web        Anthropic assistant fetcher
#   Claude-User       a person asking Claude to open a link
#   Claude-SearchBot  Anthropic search index
#   cohere-ai         Cohere
#   meta-externalagent / meta-externalfetcher   Meta AI
#   Bytespider        Doubao / TikTok
#   Amazonbot         Alexa / Rufus
#   Diffbot, Omgilibot, Timpibot, YouBot, Kangaroo Bot, ImagesiftBot
# Opt-in tokens (these are not crawlers — they are switches that decide
# whether already-crawled content may be used for AI grounding and

#   Google-Extended    Gemini / Google AI Overviews grounding
#   Applebot-Extended  Apple Intelligence
# ---------------------------------------------------------------------

# --- Ephemeral / private surfaces -----------------------------------
# /a/<askId> — an "Ask" (a time-limited request for an introduction).
# It expires, it carries a reply form, and it already ships
# <meta name="robots" content="noindex, nofollow">.
Disallow: /a/

# /i/<code> — single-use invite codes. Not content.
Disallow: /i/

# --- Interstitials with no content of their own ----------------------
# /scan is the post-QR-scan hand-off; /download is a store-redirect
# shim; /reset-password carries a one-time auth token in the URL.
Disallow: /scan
Disallow: /download
Disallow: /reset-password

# /pitch is investor material, not a product page.
Disallow: /pitch

# /app.html is the raw SPA shell that "/" is rendered from. Indexing it
# would create a duplicate of the homepage under a second URL.
Disallow: /app.html

# --- Query-string state ----------------------------------------------
# A profile URL accepts ?sid= (scan session), ?tags=, ?date= and ?loc=,
# all seeded by the in-person QR flow. Each scan mints a fresh ?sid=, so
# this is an unbounded URL space that resolves to the same page — every
# such URL already declares <link rel="canonical"> back to the bare
# /<username>. Crawling them would burn crawl budget and fire a
# server-side share_link_viewed analytics event per hit.
Disallow: /*?sid=
Disallow: /*?tags=
Disallow: /*?date=
Disallow: /*?loc=

# Campaign parameters, for the same reason and one more.
#
# The app READS utm_source / utm_campaign at launch (see
# mobile/src/lib/acquisition.ts) and the admin funnel page tells the
# founder to append them by hand, so tagged pikt.ag links exist in the
# wild whether or not we generate any. Each combination resolves to the
# same page, so they are the same unbounded space as ?sid= above.
#
# The extra reason: every crawl of one of these fires the server-side
# share_link_viewed event, which would credit a campaign with traffic a
# robot produced. Attribution that counts crawlers is worse than none.
#
# Prefix match — one line covers utm_source, utm_medium, utm_campaign,
# utm_term and utm_content. Landing FIRST, before campaign links are in
# circulation, is the point: this is cheap now and a cleanup later.
Disallow: /*?utm_
Disallow: /*&utm_

# NOT blocked: ?lang=<code>. It renders the same page in one of 19
# languages and is the only per-language URL the site has. Blocking it
# would close the door on hreflang later; rel=canonical already
# consolidates it. See llms-full.txt for the language story.
